# TensorFlow Lite Implementation on ESP32-S3 - ProxiTof Project

## Overview

This guide shows how to implement TensorFlow Lite (now LiteRT) on ESP32-S3 for the ProxiTof Near-Miss Detection System, based on best practices from OpenELab and adapted for our specific sensor fusion use case.

### 3. ML Inference Implementation

```cpp
#ifndef ML_INFERENCE_H
#define ML_INFERENCE_H

#include <stdint.h>
#include <stdbool.h>

// ProxiTof feature vector (25 features total)
typedef struct {
    // ToF sensor data (16 features)
    uint16_t tof_distances[16];  // 4x4 grid in mm
    
    // IMU orientation (3 features)
    float roll, pitch, yaw;      // degrees
    
    // IMU acceleration (3 features)
    float accel_x, accel_y, accel_z;  // m/s²
    
    // IMU gyroscope (3 features)
    float gyro_x, gyro_y, gyro_z;     // deg/s
} proxitof_features_t;

// ML inference result
typedef struct {
    float confidence;        // Near-miss probability [0.0, 1.0]
    bool near_miss_detected; // True if confidence > threshold
    uint32_t inference_time_us; // Inference latency in microseconds
} ml_result_t;

// Public API
esp_err_t ml_inference_init(void);
esp_err_t ml_inference_predict(const proxitof_features_t* features, ml_result_t* result);
void ml_inference_deinit(void);
bool ml_inference_is_ready(void);

#endif // ML_INFERENCE_H
```

**components/ml_inference/ml_inference.cpp:**
```cpp
#include "ml_inference.h"
#include "proxitof_model.h"
#include "tensorflow/lite/micro/micro_interpreter.h"
#include "tensorflow/lite/micro/micro_mutable_op_resolver.h"
#include "tensorflow/lite/micro/system_setup.h"
#include "tensorflow/lite/schema/schema_generated.h"
#include "esp_log.h"
#include "esp_timer.h"

static const char* TAG = "ML_INFERENCE";

// TensorFlow Lite Micro components
static tflite::MicroInterpreter* interpreter = nullptr;
static tflite::MicroMutableOpResolver<10> resolver;
static const tflite::Model* model = nullptr;
static TfLiteTensor* input_tensor = nullptr;
static TfLiteTensor* output_tensor = nullptr;

// Tensor arena - adjust size based on your model
constexpr int kTensorArenaSize = 50 * 1024; // 50KB
static uint8_t tensor_arena[kTensorArenaSize];

// Configuration
static float confidence_threshold = 0.7f; // 70% confidence threshold

// Feature normalization constants (from your training pipeline)
static const float TOF_MAX = 4000.0f;        // ToF sensor max range (mm)
static const float ORIENTATION_MAX = 180.0f;  // ±180 degrees
static const float ACCEL_MAX = 10.0f;        // ±10 m/s²
static const float GYRO_MAX = 500.0f;        // ±500 deg/s

static void normalize_features(const proxitof_features_t* features, float* normalized) {
    int idx = 0;
    
    // Normalize ToF distances [0, 4000mm] -> [0, 1]
    for (int i = 0; i < 16; i++) {
        normalized[idx++] = (float)features->tof_distances[i] / TOF_MAX;
    }
    
    // Normalize orientation [-180°, 180°] -> [-1, 1]
    normalized[idx++] = features->roll / ORIENTATION_MAX;
    normalized[idx++] = features->pitch / ORIENTATION_MAX;
    normalized[idx++] = features->yaw / ORIENTATION_MAX;
    
    // Normalize acceleration [-10, 10 m/s²] -> [-1, 1]
    normalized[idx++] = fmaxf(-1.0f, fminf(1.0f, features->accel_x / ACCEL_MAX));
    normalized[idx++] = fmaxf(-1.0f, fminf(1.0f, features->accel_y / ACCEL_MAX));
    normalized[idx++] = fmaxf(-1.0f, fminf(1.0f, features->accel_z / ACCEL_MAX));
    
    // Normalize gyroscope [-500, 500 deg/s] -> [-1, 1]
    normalized[idx++] = fmaxf(-1.0f, fminf(1.0f, features->gyro_x / GYRO_MAX));
    normalized[idx++] = fmaxf(-1.0f, fminf(1.0f, features->gyro_y / GYRO_MAX));
    normalized[idx++] = fmaxf(-1.0f, fminf(1.0f, features->gyro_z / GYRO_MAX));
}

esp_err_t ml_inference_init(void) {
    ESP_LOGI(TAG, "Initializing TensorFlow Lite Micro");
    
    // Initialize TensorFlow Lite Micro
    tflite::InitializeTarget();
    
    // Load model from flash
    model = tflite::GetModel(proxitof_model_data);
    if (model->version() != TFLITE_SCHEMA_VERSION) {
        ESP_LOGE(TAG, "Model schema version %d not supported. Supported version is %d",
                 model->version(), TFLITE_SCHEMA_VERSION);
        return ESP_FAIL;
    }
    
    // Add required operations (adjust based on your model)
    resolver.AddFullyConnected();
    resolver.AddRelu();
    resolver.AddSigmoid();
    resolver.AddSoftmax();
    // ESP-NN optimizations are automatically applied
    
    // Create interpreter
    static tflite::MicroInterpreter static_interpreter(
        model, resolver, tensor_arena, kTensorArenaSize);
    interpreter = &static_interpreter;
    
    // Allocate tensors
    TfLiteStatus allocate_status = interpreter->AllocateTensors();
    if (allocate_status != kTfLiteOk) {
        ESP_LOGE(TAG, "AllocateTensors() failed");
        return ESP_FAIL;
    }
    
    // Get input and output tensors
    input_tensor = interpreter->input(0);
    output_tensor = interpreter->output(0);
    
    // Verify tensor shapes
    if (input_tensor->dims->size != 2 || input_tensor->dims->data[1] != 25) {
        ESP_LOGE(TAG, "Input tensor shape mismatch. Expected [1, 25], got [%d, %d]",
                 input_tensor->dims->data[0], input_tensor->dims->data[1]);
        return ESP_FAIL;
    }
    
    ESP_LOGI(TAG, "TensorFlow Lite Micro initialized successfully");
    ESP_LOGI(TAG, "Model size: %d bytes", proxitof_model_len);
    ESP_LOGI(TAG, "Tensor arena size: %d bytes", kTensorArenaSize);
    ESP_LOGI(TAG, "Input shape: [%d, %d]", input_tensor->dims->data[0], input_tensor->dims->data[1]);
    ESP_LOGI(TAG, "Output shape: [%d, %d]", output_tensor->dims->data[0], output_tensor->dims->data[1]);
    
    return ESP_OK;
}

esp_err_t ml_inference_predict(const proxitof_features_t* features, ml_result_t* result) {
    if (!interpreter || !input_tensor || !output_tensor) {
        ESP_LOGE(TAG, "ML inference not initialized");
        return ESP_FAIL;
    }
    
    uint64_t start_time = esp_timer_get_time();
    
    // Normalize input features
    float normalized_features[25];
    normalize_features(features, normalized_features);
    
    // Copy normalized features to input tensor
    float* input_data = input_tensor->data.f;
    for (int i = 0; i < 25; i++) {
        input_data[i] = normalized_features[i];
    }
    
    // Run inference
    TfLiteStatus invoke_status = interpreter->Invoke();
    if (invoke_status != kTfLiteOk) {
        ESP_LOGE(TAG, "Invoke() failed");
        return ESP_FAIL;
    }
    
    // Get prediction result
    float* output_data = output_tensor->data.f;
    result->confidence = output_data[0];
    result->near_miss_detected = (result->confidence > confidence_threshold);
    result->inference_time_us = esp_timer_get_time() - start_time;
    
    // Log performance warning if inference is slow
    if (result->inference_time_us > 50000) { // 50ms threshold
        ESP_LOGW(TAG, "Inference latency high: %lu us", result->inference_time_us);
    }
    
    return ESP_OK;
}

void ml_inference_deinit(void) {
    interpreter = nullptr;
    input_tensor = nullptr;
    output_tensor = nullptr;
    ESP_LOGI(TAG, "TensorFlow Lite Micro deinitialized");
}

bool ml_inference_is_ready(void) {
    return (interpreter != nullptr && input_tensor != nullptr && output_tensor != nullptr);
}
```

## Integration with ProxiTof Main Application

**main/main.cpp integration:**
```cpp
#include "ml_inference.h"
#include "sensor_manager.h"
#include "data_processor.h"

void ml_task(void* pvParameters) {
    // Initialize ML inference
    if (ml_inference_init() != ESP_OK) {
        ESP_LOGE(TAG, "Failed to initialize ML inference");
        vTaskDelete(NULL);
        return;
    }
    
    proxitof_features_t features;
    ml_result_t ml_result;
    
    while (1) {
        // Get synchronized sensor data from data processor
        if (get_processed_features(&features) == ESP_OK) {
            
            // Run ML inference
            if (ml_inference_predict(&features, &ml_result) == ESP_OK) {
                
                // Log ML detection
                if (ml_result.near_miss_detected) {
                    ESP_LOGW(TAG, "Near-miss detected! Confidence: %.3f, Latency: %lu us",
                             ml_result.confidence, ml_result.inference_time_us);
                    
                    // Trigger LED alert
                    set_led_state(LED_ML_NEAR_MISS);
                    
                    // Log to SD card with auto-flag
                    log_event_with_ml_flag(ml_result.confidence);
                }
                
                // Performance monitoring
                if (ml_result.inference_time_us > 50000) {
                    ESP_LOGW(TAG, "ML inference slow: %lu us", ml_result.inference_time_us);
                }
            }
        }
        
        // Run inference at 60Hz (every 16.67ms)
        vTaskDelay(pdMS_TO_TICKS(17));
    }
}

void app_main(void) {
    // Initialize other components...
    
    // Create ML inference task
    xTaskCreatePinnedToCore(
        ml_task,
        "ml_task",
        8192,  // Stack size
        NULL,
        5,     // Priority (high for real-time inference)
        NULL,
        1      // Core 1 (Core 0 for sensors)
    );
}
```

## Performance Targets for ProxiTof

**Memory Usage:**
- Model size: <100KB flash
- Tensor arena: ~50KB RAM
- Total ML RAM: <80KB

**Timing:**
- Inference latency: <50ms (target <20ms)
- Frequency: 60Hz (every 16.67ms)
- Sensor-to-decision: <100ms total

**Accuracy:**
- Precision: >85% (minimize false positives)
- Recall: >90% (catch near-miss events)
- Confidence threshold: 0.7 (adjustable via config)
